namespace Monetization.Runtime.Analytics
{
    /// <summary>
    /// Metica analytics interface for the Shooter genre.
    /// Retrieve via <c>AnalyticsManager.Game</c> using this interface as the type parameter.
    /// </summary>
    public interface IShooterAnalytics : IGDMeticaAnalytics
    {
        /// <summary>Logs a levelStarted event when the player enters a shooter level.</summary>
        void LevelStarted(ShooterLevelStarted data);

        /// <summary>Logs a levelFinished event when a shooter level ends (win or fail).</summary>
        void LevelFinished(ShooterLevelFinished data);

        /// <summary>Logs an earnAsset event when the player receives an in-game asset.</summary>
        void EarnAsset(ShooterEarnAsset data);

        /// <summary>Logs a spendAsset event when the player spends an in-game asset.</summary>
        void SpendAsset(ShooterSpendAsset data);
    }
}
