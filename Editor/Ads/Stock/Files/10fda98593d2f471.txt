using System.Collections.Generic;

namespace Monetization.Runtime.Analytics
{
    // LevelData fields: Level=R, LevelType=O, LevelId=O
    public class PuzzleSuccessLevelStarted : LevelData
    {
        // ── Optional — null means not set, will not appear in payload ─────────
        public string LevelName { get; set; }

        public PuzzleSuccessLevelStarted(int level)
        {
            Level = level;
        }
    }

    public class PuzzleSuccessLevelFinished
    {
        // ── Required ──────────────────────────────────────────────────────────
        public int    Level            { get; }
        public string LevelStatus      { get; }
        public float  PlayTime         { get; }
        public int    AttemptCount     { get; }
        public int                       UsedBoosterCount { get; }
        public Dictionary<string, int>   UsedBoosters     { get; }
        public Dictionary<string, float> Balances         { get; }
        public Dictionary<string, float> BalanceBoosters  { get; }

        // ── Optional — null/no value means not set, will not appear in payload ─
        public string LevelName   { get; set; }
        public string LevelType   { get; set; }
        public int?   PlayonCount { get; set; }

        public PuzzleSuccessLevelFinished(int level, string levelStatus, float playTime,
            int attemptCount, int usedBoosterCount, Dictionary<string, int> usedBoosters,
            Dictionary<string, float> balances, Dictionary<string, float> balanceBoosters)
        {
            Level            = level;
            LevelStatus      = levelStatus;
            PlayTime         = playTime;
            AttemptCount     = attemptCount;
            UsedBoosterCount = usedBoosterCount;
            UsedBoosters     = usedBoosters;
            Balances         = balances;
            BalanceBoosters  = balanceBoosters;
        }
    }

    // AssetData fields: Level=R, Asset=R, Amount=R, Source=R
    // LevelName=O, LevelType=O
    public class PuzzleSuccessEarnAsset : AssetData
    {
        // ── Required ──────────────────────────────────────────────────────────
        public string Type    { get; }
        public float  Balance { get; }

        // ── Optional ─────────────────────────────────────────────────────────
        public string LevelName { get; set; }

        public PuzzleSuccessEarnAsset(int level, string asset, float amount, string source,
            string type, float balance)
            : base(level, asset, amount, source)
        {
            Type    = type;
            Balance = balance;
        }
    }

    // AssetData fields: Level=R, Asset=R, Amount=R, Source=R
    // LevelName=O, LevelType=O
    public class PuzzleSuccessSpendAsset : AssetData
    {
        // ── Required ──────────────────────────────────────────────────────────
        public float Balance { get; }

        // ── Optional ─────────────────────────────────────────────────────────
        public string LevelName { get; set; }

        public PuzzleSuccessSpendAsset(int level, string asset, float amount, string source,
            float balance)
            : base(level, asset, amount, source)
        {
            Balance = balance;
        }
    }
}
