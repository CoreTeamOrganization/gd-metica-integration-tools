using UnityEngine;

namespace Monetization.Runtime.Analytics
{
    [CreateAssetMenu(fileName = "PuzzleSuccessFailAnalytics", menuName = "GDMonetization/MeticaAnalytics/ScriptSO/PuzzleSuccessFail")]
    public class PuzzleSuccessFailAnalytics : GDMeticaAnalytics, IPuzzleSuccessFailAnalytics
    {
        public void LevelStarted(PuzzleSuccessFailLevelStarted data)
        {
            var payload = CreateBaseEvent();
            payload["level"] = data.Level.Value;

            if (data.LevelType != null)       payload["level_type"]       = data.LevelType;
            if (data.LevelDifficulty != null) payload["level_difficulty"] = data.LevelDifficulty;
            if (data.LevelId.HasValue)        payload["level_id"]         = data.LevelId.Value;

            LogCustomEvent("levelStarted", payload);
        }

        public void LevelFinished(PuzzleSuccessFailLevelFinished data)
        {
            var payload = CreateBaseEvent();
            payload["level"]                = data.Level.Value;
            payload["level_status"]         = data.LevelStatus;
            payload["reason"]               = data.Reason;
            payload["play_time"]            = data.PlayTime;
            payload["total_objectives"]     = data.TotalObjectives;
            payload["remaining_objectives"] = data.RemainingObjectives;
            payload["attempt_count"]        = data.AttemptCount;
            payload["playon_count"]         = data.PlayonCount;
            payload["used_booster_count"] = data.UsedBoosterCount;
            foreach (var kvp in data.UsedBoosters)
                payload["used_booster_" + kvp.Key] = kvp.Value;
            foreach (var kvp in data.Balances)
                payload["balance_" + kvp.Key] = kvp.Value;
            foreach (var kvp in data.BalanceBoosters)
                payload["balance_booster_" + kvp.Key] = kvp.Value;

            if (data.LevelType != null)               payload["level_type"]             = data.LevelType;
            if (data.LevelDifficulty != null)         payload["level_difficulty"]       = data.LevelDifficulty;
            if (data.LevelId.HasValue)                payload["level_id"]               = data.LevelId.Value;
            if (data.TotalTime.HasValue)              payload["total_time"]             = data.TotalTime.Value;
            if (data.RemainingTime.HasValue)          payload["remaining_time"]         = data.RemainingTime.Value;
            if (data.ExtraTime.HasValue)              payload["extra_time"]             = data.ExtraTime.Value;
            if (data.TotalMoves.HasValue)             payload["total_moves"]            = data.TotalMoves.Value;
            if (data.RemainingMoves.HasValue)         payload["remaining_moves"]        = data.RemainingMoves.Value;
            if (data.ExtraMoves.HasValue)             payload["extra_moves"]            = data.ExtraMoves.Value;
            if (data.RemainingLife.HasValue)          payload["remaining_life"]         = data.RemainingLife.Value;
            if (data.PlayonRewardedCount.HasValue)    payload["playon_rewarded_count"]  = data.PlayonRewardedCount.Value;

            LogCustomEvent("levelFinished", payload);
        }

        public void EarnAsset(PuzzleSuccessFailEarnAsset data)
        {
            var payload = CreateBaseEvent();
            payload["level"]   = data.Level.Value;
            payload["asset"]   = data.Asset;
            payload["amount"]  = data.Amount;
            payload["source"]  = data.Source;
            payload["type"]    = data.Type;
            payload["balance"] = data.Balance;

            if (data.LevelType != null)       payload["level_type"]       = data.LevelType;
            if (data.LevelDifficulty != null) payload["level_difficulty"] = data.LevelDifficulty;
            if (data.LevelId.HasValue)        payload["level_id"]         = data.LevelId.Value;

            LogCustomEvent("earnAsset", payload);
        }

        public void SpendAsset(PuzzleSuccessFailSpendAsset data)
        {
            var payload = CreateBaseEvent();
            payload["level"]   = data.Level.Value;
            payload["asset"]   = data.Asset;
            payload["amount"]  = data.Amount;
            payload["source"]  = data.Source;
            payload["balance"] = data.Balance;

            if (data.LevelType != null)       payload["level_type"]       = data.LevelType;
            if (data.LevelDifficulty != null) payload["level_difficulty"] = data.LevelDifficulty;
            if (data.LevelId.HasValue)        payload["level_id"]         = data.LevelId.Value;

            LogCustomEvent("spendAsset", payload);
        }
    }
}
