using System.Collections.Generic;
using System.Text.RegularExpressions;
using Metica;
using Monetization.Runtime.Sdk;

namespace Monetization.Runtime.Analytics
{
    internal sealed class MeticaAnalyticsNetwork : IAnalyticsInitialization, IAnalyticsEventsService
    {
        #region vars
        private static readonly Regex CamelCaseRegex = new(@"^[a-z][a-zA-Z0-9]*$", RegexOptions.Compiled);
        private static readonly Regex SplitRegex = new(@"[\s_\-]+|(?<=[a-z])(?=[A-Z])", RegexOptions.Compiled);
        private static readonly Dictionary<string, string> EventNameCache = new();
        #endregion

        #region IAnalyticsInitialization
        public bool IsInitialized => MeticaInitializer.IsInitialized;
        public void Initialize()
        {
        }
        #endregion

        #region IAnalyticsEventsService
        public void SendEvent(string eventName)
        {
        }

        public void SendEvent(string eventName, Dictionary<string, object> parameters)
        {
            var payload = new Dictionary<string, object>
            {
                { "adid",      AdjustAnalyticsNetwork.AdId },
                { "app_token", AdjustAnalyticsNetwork.AppToken }
            };

            if (parameters != null)
                foreach (var kvp in parameters)
                    payload[kvp.Key] = kvp.Value;

            MeticaSdk.Analytics.LogCustomEvent(GetSafeEventName(eventName), payload);
        }
        #endregion

        #region Helpers

        private static string GetSafeEventName(string input)
        {
            if (EventNameCache.TryGetValue(input, out var cached))
                return cached;

            IsCamelCase(input, out var safe);
            EventNameCache[input] = safe;
            return safe;
        }

        /// <summary>
        /// Returns true if <paramref name="input"/> is already camelCase.
        /// Otherwise converts it to camelCase and returns the result via <paramref name="output"/>.
        /// </summary>
        private static bool IsCamelCase(string input, out string output)
        {
            if (string.IsNullOrEmpty(input) || CamelCaseRegex.IsMatch(input))
            {
                output = input;
                return true;
            }

            output = ToCamelCase(input);
            return false;
        }

        private static string ToCamelCase(string input)
        {
            // Split on spaces/underscores/hyphens or at PascalCase word boundaries, then remove empty entries
            var words = SplitRegex.Split(input);

            var result = string.Empty;
            var isFirst = true;
            foreach (var word in words)
            {
                if (word.Length == 0) continue;

                if (isFirst)
                {
                    result += word.ToLower();
                    isFirst = false;
                }
                else
                {
                    result += char.ToUpper(word[0]) + word[1..].ToLower();
                }
            }

            return result;
        }
        #endregion


    }
}
