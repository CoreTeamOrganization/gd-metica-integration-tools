namespace Monetization.Runtime.Analytics
{
    public class SimulationJobDayCompleted
    {
        // ── Required ──────────────────────────────────────────────────────────
        public int   Day               { get; }
        public float PlayTime          { get; }
        public float CashSpent         { get; }
        public float CashEarned        { get; }
        public float CashBalance       { get; }
        public float EnergySpent       { get; }
        public float EnergyEarned      { get; }
        public float EnergyBalance     { get; }
        public int   RewardedCount     { get; }
        public int   InterstitialCount { get; }
        public int   IapCount          { get; }

        // ── Optional — null/no value means not set, will not appear in payload ─
        public int? Level              { get; set; }
        public int? Hour               { get; set; }
        public int? CustomersProcessed { get; set; }
        public int? ItemsCollected     { get; set; }
        public int? ItemsSold          { get; set; }

        public SimulationJobDayCompleted(int day, float playTime,
            float cashSpent, float cashEarned, float cashBalance,
            float energySpent, float energyEarned, float energyBalance,
            int rewardedCount, int interstitialCount, int iapCount)
        {
            Day               = day;
            PlayTime          = playTime;
            CashSpent         = cashSpent;
            CashEarned        = cashEarned;
            CashBalance       = cashBalance;
            EnergySpent       = energySpent;
            EnergyEarned      = energyEarned;
            EnergyBalance     = energyBalance;
            RewardedCount     = rewardedCount;
            InterstitialCount = interstitialCount;
            IapCount          = iapCount;
        }
    }
