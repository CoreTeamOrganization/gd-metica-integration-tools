using System;
using System.Threading.Tasks;
using UnityEngine;
using Monetization.Runtime.Logger;
using Monetization.Runtime.Consent;
using Monetization.Runtime.Configurations;
using Monetization.Runtime.Sdk;
using Metica.Ads;
using Metica;
using Monetization.Runtime.Analytics;
using System.Collections.Generic;

namespace Monetization.Runtime.Ads
{
    internal sealed class AdNetworkMetica : IAsyncAdNetworkService
    {
        public static string GetUserGroup
        {
            get
            {
                if (SmartFloorsResult != null)
                {
                    if (SmartFloorsResult.IsSuccess)
                    {
                        return SmartFloorsResult.UserGroup == MeticaUserGroup.TRIAL ? "Trial" : "Holdout";
                    }

                    return "Holdout";
                }
                return "Initializing";
            }
        }
        static MeticaSmartFloors SmartFloorsResult;
        public async Task InitializeAsync(string MAX_SDK_KEY, Action onInitialized)
        {
            SmartFloorsResult = null;
            Message.Log(Tag.Metica, $"Initializing Metica SDK v{MeticaSdk.Version}...");

            var meticaSO = Resources.Load<MeticaConfiguration>(MonetizationConfigurationsPath.Metica);

            MeticaSdk.SetLogEnabled(true);

            var config = new MeticaInitConfig(meticaSO.ApiKey, meticaSO.AppID, "");
            var mediationInfo = new MeticaMediationInfo(MeticaMediationInfo.MeticaMediationType.MAX, MAX_SDK_KEY);

            // Measure Metica SDK initialization time
            float initStartTime = Time.realtimeSinceStartup;

            // 3️⃣ Await the SDK initialization
            var result = await MeticaSdk.InitializeAsync(config, mediationInfo);

            float initDuration = Time.realtimeSinceStartup - initStartTime;
            Message.Log(Tag.Metica, $"Metica SDK initialization took {initDuration:F3} seconds.");

            ConsentManager.AddAndUpdateConsentService(new MeticaConsentSettings());

            SmartFloorsResult = result.SmartFloors;

            string userGroup = result.SmartFloors.IsSuccess ? result.SmartFloors.UserGroup.ToString() : "HOLDOUT_WITH_ERROR";
            // 5️⃣ Log success and invoke callback
            Message.Log(Tag.Metica, $"Metica initialization completed: {result.SmartFloors.IsSuccess}");
            Message.Log(Tag.Metica, $"Current user is part of {userGroup}");
            Message.Log(Tag.Metica, $"Metica SDK version: {MeticaSdk.Version}");

            SendDetailsEvent(userGroup);

            Message.Log(Tag.Metica, "Successfully initialized");

            // float delay = 2f;
            // Message.Log(Tag.Metica, $"Invoking callback after {delay} seconds");

            // await Task.Delay((int)(delay * 1000));
            onInitialized?.Invoke();
       }

        void SendDetailsEvent(string metica_user_group)
        {
            Message.Log(Tag.Metica, $"Sending Metica user-group and sdk-version to Analytics: {metica_user_group}");

            var param1 = new Dictionary<string, object> { { "UserGroup", metica_user_group } };
            var param2 = new Dictionary<string, object> { { "Metica", param1 } };
            AnalyticsManager.SendEvent("Monetization", param2);

            param1 = new Dictionary<string, object> { { "SdkVersion", MeticaSdk.Version } };
            param2 = new Dictionary<string, object> { { "Metica", param1 } };

            AnalyticsManager.SendEvent("Monetization", param2);
        }
    }
}
