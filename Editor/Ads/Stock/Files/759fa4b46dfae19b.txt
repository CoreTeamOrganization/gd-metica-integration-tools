using System.IO;
using UnityEditor;
using UnityEditor.Callbacks;
using UnityEngine;

public class UnityPostProcess
{
    [PostProcessBuild(45)]
    public static void OnPostProcessBuild(BuildTarget buildTarget, string buildPath)
    {
#if UNITY_IOS
        if (buildTarget != BuildTarget.iOS) return;

        string podfilePath = Path.Combine(buildPath, "Podfile");
        PatchPodfile(podfilePath);
#endif
    }

    private static void PatchPodfile(string podfilePath)
    {
        if (!File.Exists(podfilePath))
        {
            Debug.LogError("Podfile not found at: " + podfilePath);
            return;
        }

        string podfile = File.ReadAllText(podfilePath);

        if (podfile.Contains("BUILD_LIBRARY_FOR_DISTRIBUTION"))
        {
            Debug.Log("Podfile already patched.");
            return;
        }

        string postInstallHook = @"
post_install do |installer|
  installer.pods_project.targets.each do |target|
    target.build_configurations.each do |config|
      if target.name.include?('AppMetrica') || target.name.include?('Yandex')
        config.build_settings['BUILD_LIBRARY_FOR_DISTRIBUTION'] = 'NO'
      end
    end
  end
end
";

        podfile += postInstallHook;

        File.WriteAllText(podfilePath, podfile);

        Debug.Log("Podfile patched successfully.");
    }
}
