namespace Monetization.Runtime.Analytics
{
    /// <summary>
    /// Metica analytics interface for the Puzzle (Success &amp; Fail) genre.
    /// Retrieve via <c>AnalyticsManager.Game</c> using this interface as the type parameter.
    /// </summary>
    public interface IPuzzleSuccessFailAnalytics : IGDMeticaAnalytics
    {
        /// <summary>Logs a levelStarted event when the player enters a puzzle level.</summary>
        void LevelStarted(PuzzleSuccessFailLevelStarted data);

        /// <summary>Logs a levelFinished event when a puzzle level ends (win or fail).</summary>
        void LevelFinished(PuzzleSuccessFailLevelFinished data);

        /// <summary>Logs an earnAsset event when the player receives an in-game asset.</summary>
        void EarnAsset(PuzzleSuccessFailEarnAsset data);

        /// <summary>Logs a spendAsset event when the player spends an in-game asset.</summary>
        void SpendAsset(PuzzleSuccessFailSpendAsset data);
    }
}
