using Metica;
using Metica.Ads;
using UnityEngine;

namespace MeticaIntegration
{
    public sealed class MeticaMRec : MRecAdUnit
    {
        public override void Initialize(string adUnitId)
        {
            this.adUnitId = adUnitId;
            if (IsAdUnitEmpty)
            {
                MeticaAdsLog.LogError("Initialize: MRec ad unit is empty");
                return;
            }
            ListenEvents();
            CreateMRec();
        }

        public void ListenEvents()
        {
            MeticaAdsCallbacks.Mrec.OnAdLoadSuccess += MRec_OnAdLoadedEvent;
            MeticaAdsCallbacks.Mrec.OnAdLoadFailed += MRec_OnAdLoadFailedEvent;
            MeticaAdsCallbacks.Mrec.OnAdRevenuePaid += MRec_OnAdRevenuePaidEvent;
            MeticaAdsCallbacks.Mrec.OnAdClicked += MRec_OnAdClickedEvent;
        }

        public void CreateMRec()
        {
            MeticaSdk.Ads.CreateMrec(adUnitId, HasOffset
                ? new MeticaAdViewConfiguration(OffsetX(), OffsetY())
                : new MeticaAdViewConfiguration(ConvertToMeticaMRecPos()));
            SetMRecAutoRefreshStatus(AutoRefresh.Enabled);
            MeticaAdsLog.Log("MRec IS CREATED & LOADING");
            isLoading = true;
        }

        public override bool HasMRec()
        {
            if (!hasMrec)
            {
                LoadMRec();
                return false;
            }

            return true;
        }

        public override void LoadMRec()
        {
            if (isLoading)
            {
                MeticaAdsLog.LogWarning("MRec IS ALREADY LOADING");
                return;
            }

            isLoading = true;
            MeticaAdsLog.LogWarning("MRec IS LOADING");
            MeticaSdk.Ads.LoadMrec(adUnitId);
        }

        public override void ShowMRec()
        {
            MeticaAdsLog.Log("MRec IS SHOWING");
            MeticaSdk.Ads.ShowMrec(adUnitId);
        }

        public override void HideMRec()
        {
            MeticaAdsLog.Log("MRec IS HIDDEN");
            MeticaSdk.Ads.HideMrec(adUnitId);
        }

        public override void DestroyMRec()
        {
            MeticaAdsLog.Log("MRec IS DESTROYED");
            HideMRec();
        }

        public override void RepositionMRec(MRecPosition newPosition, Vector2 offset = default)
        {
            bannerInfo.MrecPosition = newPosition;
            bannerInfo.MrecOffset = offset;
            if (IsAdUnitEmpty) return;

            if (HasOffset)
            {
                MeticaAdsLog.Log($"MRec is repositioned to {newPosition} shifted by {offset} dp");
                MeticaSdk.Ads.UpdateMrecPositionCoordinates(adUnitId, OffsetX(), OffsetY());
            }
            else
            {
                MeticaAdsLog.Log($"MRec is repositioned to {newPosition}");
                MeticaSdk.Ads.UpdateMrecPosition(adUnitId, ConvertToMeticaMRecPos());
            }
        }

        // ── Custom position: an anchor shifted by an offset ────────────────────
        //
        // Metica places an MREC either at one of nine anchors or at x/y — dp on Android,
        // points on iOS, from the top-left. An offset needs x/y, so the anchor is worked out
        // here from the screen size in dp and the MREC's size. An MREC is always 300 × 250:
        // neither Metica nor MAX can resize one.

        private const float Width = 300f;
        private const float Height = 250f;

        private bool HasOffset => bannerInfo.MrecOffset != Vector2.zero;

        private double OffsetX()
        {
            var screen = Screen.width / Density;
            var x = ColumnOf(bannerInfo.MrecPosition) * (screen - Width) / 2f + bannerInfo.MrecOffset.x;
            return Mathf.Clamp(x, 0f, Mathf.Max(0f, screen - Width));
        }

        private double OffsetY()
        {
            var screen = Screen.height / Density;
            var y = RowOf(bannerInfo.MrecPosition) * (screen - Height) / 2f + bannerInfo.MrecOffset.y;
            return Mathf.Clamp(y, 0f, Mathf.Max(0f, screen - Height));
        }

        /// <summary>0 left, 1 center, 2 right.</summary>
        private static int ColumnOf(MRecPosition position)
        {
            switch (position)
            {
                case MRecPosition.TopLeft:
                case MRecPosition.CenterLeft:
                case MRecPosition.BottomLeft:
                    return 0;
                case MRecPosition.TopRight:
                case MRecPosition.CenterRight:
                case MRecPosition.BottomRight:
                    return 2;
                default:
                    return 1;
            }
        }

        /// <summary>0 top, 1 center, 2 bottom.</summary>
        private static int RowOf(MRecPosition position)
        {
            switch (position)
            {
                case MRecPosition.Top:
                case MRecPosition.TopLeft:
                case MRecPosition.TopRight:
                    return 0;
                case MRecPosition.Bottom:
                case MRecPosition.BottomLeft:
                case MRecPosition.BottomRight:
                    return 2;
                default:
                    return 1;
            }
        }

        private static float _density;

        /// <summary>Screen pixels per dp (per point on iOS).</summary>
        private static float Density
        {
            get
            {
                if (_density > 0f) return _density;

#if UNITY_ANDROID && !UNITY_EDITOR
                // The exact value Android lays the MREC out with.
                try
                {
                    using (var player = new AndroidJavaClass("com.unity3d.player.UnityPlayer"))
                    using (var activity = player.GetStatic<AndroidJavaObject>("currentActivity"))
                    using (var resources = activity.Call<AndroidJavaObject>("getResources"))
                    using (var metrics = resources.Call<AndroidJavaObject>("getDisplayMetrics"))
                        _density = metrics.Get<float>("density");
                }
                catch (System.Exception e)
                {
                    MeticaAdsLog.LogWarning($"MRec: could not read the screen density ({e.Message}) — estimating it");
                }
#endif
                if (_density <= 0f)
                {
                    // An estimate: 160 dpi is 1 dp per pixel on Android, 163 is 1 point on iOS.
                    var dpi = Screen.dpi > 0f ? Screen.dpi : 160f;
#if UNITY_IOS
                    _density = dpi / 163f;
#else
                    _density = dpi / 160f;
#endif
                }

                return _density;
            }
        }

        MeticaAdViewPosition ConvertToMeticaMRecPos()
        {
            switch (bannerInfo.MrecPosition)
            {
                case MRecPosition.Top:
                    return MeticaAdViewPosition.TopCenter;
                case MRecPosition.TopLeft:
                    return MeticaAdViewPosition.TopLeft;
                case MRecPosition.TopRight:
                    return MeticaAdViewPosition.TopRight;
                case MRecPosition.Center:
                    return MeticaAdViewPosition.Centered;
                case MRecPosition.CenterLeft:
                    return MeticaAdViewPosition.CenterLeft;
                case MRecPosition.CenterRight:
                    return MeticaAdViewPosition.CenterRight;
                case MRecPosition.Bottom:
                    return MeticaAdViewPosition.BottomCenter;
                case MRecPosition.BottomLeft:
                    return MeticaAdViewPosition.BottomLeft;
                case MRecPosition.BottomRight:
                    return MeticaAdViewPosition.BottomRight;
                default:
                    return MeticaAdViewPosition.BottomCenter;
            }
        }

        #region Callbacks

        private void MRec_OnAdLoadedEvent(MeticaAd meticaAd)
        {
            ThreadDispatcher.Enqueue(() =>
            {
                isLoading = false;
                hasMrec = true;
                MeticaAdsLog.Log("MRec IS LOADED");
                SetMRecAutoRefreshStatus(AutoRefresh.Enabled);
                if (MeticaAdsManager.MRecStatus)
                {
                    MeticaAdsManager.ShowMRec();
                    return;
                }

                HideMRec();
            });
        }

        private void MRec_OnAdLoadFailedEvent(MeticaAdError meticaAdError)
        {
            ThreadDispatcher.Enqueue(() =>
            {
                isLoading = false;
                hasMrec = false;
                SetMRecAutoRefreshStatus(AutoRefresh.Disabled);
                HideMRec();
                MeticaAdsLog.LogError("MRec failed to load");

                DelayedActionManager.Add(() =>
                {
                    if (MeticaAdsManager.MRecStatus)
                    {
                        MeticaAdsManager.ShowMRec();
                    }
                }, 3f, "Retry Metica MRec");
            });
        }

        private void MRec_OnAdRevenuePaidEvent(MeticaAd meticaAd)
        {
            double revenue = meticaAd.revenue ?? 0.0;
            string networkName = meticaAd.networkName ?? "Unknown";
            string adUnitIdentifier = meticaAd.adUnitId;

            AdRevenueInfo revInfo = new AdRevenueInfo(AdFormat.MRec, AdNetworkMetica.Platform, networkName,
                adUnitIdentifier, "USD", revenue, "none", AdNetworkMetica.RevenuePayload);

            MeticaAdsLog.Log(
                $"MRec revenue paid - AdUnit: {adUnitIdentifier}, Revenue: {revenue}, Network: {networkName}");

            // On the game screen, so reported on the main thread, like the rest of its callbacks.
            ThreadDispatcher.Enqueue(() => MeticaAdsHooks.ReportRevenue(revInfo));
        }

        private void MRec_OnAdClickedEvent(MeticaAd meticaAd)
        {
            ThreadDispatcher.Enqueue(MeticaAdsHooks.ReportAdClicked);
            MeticaAdsLog.Log("MRec ad clicked");
        }

        #endregion

        #region AutoRefreshStatus

        private AutoRefresh refreshStatus = AutoRefresh.NotSet;

        void SetMRecAutoRefreshStatus(AutoRefresh value)
        {
            if (refreshStatus == AutoRefresh.NotSet || refreshStatus != value)
            {
                refreshStatus = value;
                if (refreshStatus == AutoRefresh.Enabled)
                {
                    MeticaAdsLog.LogWarning("MRec REFRESH : " + value.ToString());
                    MeticaSdk.Ads.StartMrecAutoRefresh(adUnitId);
                }
                else
                {
                    MeticaAdsLog.LogWarning("MRec REFRESH : " + value.ToString());
                    MeticaSdk.Ads.StopMrecAutoRefresh(adUnitId);
                }
            }
        }

        public enum AutoRefresh
        {
            NotSet,
            Enabled,
            Disabled
        }

        #endregion
    }
}
