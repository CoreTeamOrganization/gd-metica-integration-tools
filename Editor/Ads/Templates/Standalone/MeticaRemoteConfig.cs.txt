using System;
using System.Collections.Generic;
using UnityEngine;

namespace MeticaIntegration
{
    /// <summary>
    /// The remote switch that decides whether Metica runs at all — a kill switch you can
    /// throw from your backend without shipping a build, and the A/B split between Metica and
    /// AppLovin MAX.
    ///
    /// <para>The awkward part, and the reason this is not just a bool: remote config arrives
    /// well after the point where Metica has to be initialized. Waiting for the fetch would
    /// delay every ad in the session, and initializing first would make the switch useless.
    /// So the value is <b>persisted</b>: <see cref="Apply(string)"/> writes what your remote
    /// config says, and the next session reads it at boot. The same trade the GD Monetization
    /// SDK makes — a change takes effect on the session after the one that received it.</para>
    ///
    /// <para>What is stored is the main ad network, a <see cref="global::MainAdNetwork"/>
    /// saved through <see cref="MainAdNetworkPref"/>: Unassigned until a fetch carrying the
    /// switch lands, then what it says — MAX, Metica, or Unassigned again. Remote config
    /// carries those same three words, and nothing else is ever stored. Users still Unassigned
    /// get the config's default network, but are reported as Unassigned, so a Metica-vs-MAX
    /// comparison can leave them out instead of counting them as MAX.</para>
    ///
    /// <para>Metica is off by default: MeticaAdsConfig has <c>UseRemoteSwitch</c> on and
    /// <c>DefaultUseMetica</c> off, so Metica stays off until your remote config calls
    /// <see cref="Apply(string)"/> with "Metica" — and then from the next session. A game that
    /// never calls Apply never runs Metica. Unticking <c>UseRemoteSwitch</c> skips all of this
    /// and Metica always initializes.</para>
    /// </summary>
    public static class MeticaRemoteConfig
    {
        private static bool _reported;
        private static bool _logged;

        /// <summary>Once per session, also in the Editor with domain reload off.</summary>
        [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.SubsystemRegistration)]
        private static void ResetSession()
        {
            _reported = false;
            _logged = false;
        }

        /// <summary>True once remote config has said anything on this device — an explicit Unassigned included.</summary>
        public static bool HasStoredValue => MainAdNetworkPref.HasValue;

        /// <summary>What this device has stored; Unassigned until remote config has said otherwise.</summary>
        public static MainAdNetwork MainAdNetwork => MainAdNetworkPref.Get();

        /// <summary>
        /// Records what your remote config says the main ad network is: "Unassigned", "MAX" or
        /// "Metica", in any casing. Call it whenever a fetch completes — the value is for the
        /// next session, not this one.
        ///
        /// <para>"Unassigned" takes the device back out of the test. A value that is none of
        /// the three is refused: nothing changes, and an error is logged. An empty value (your
        /// remote config has no such key for this user) changes nothing either, so a user
        /// outside the test stays as they are.</para>
        /// </summary>
        /// <returns>True when the value was one of the three and is now stored.</returns>
        public static bool Apply(string mainAdNetwork)
        {
            if (string.IsNullOrWhiteSpace(mainAdNetwork))
            {
                MeticaAdsLog.Log($"Remote has no MainAdNetwork, MainAdNetwork stays {MainAdNetwork}");
                return false;
            }

            if (!MainAdNetworkPref.TryParse(mainAdNetwork, out var network))
            {
                MeticaAdsLog.LogError($"Remote MainAdNetwork \"{mainAdNetwork}\" is not one of " +
                                      $"{string.Join(", ", Enum.GetNames(typeof(MainAdNetwork)))}, " +
                                      $"MainAdNetwork stays {MainAdNetwork}");
                return false;
            }

            Store(network, null);
            return true;
        }

        /// <summary>Records a network the game has already parsed, as <see cref="Apply(string)"/> does.</summary>
        public static void Apply(MainAdNetwork network) => Store(network, null);

        /// <summary>
        /// The earlier switch, a bool: true stores Metica, false stores MAX — for a remote
        /// config that has a true / false flag. Only call it when the fetched config actually
        /// has the flag: a user outside the test should stay Unassigned.
        /// </summary>
        public static void Apply(bool useMetica) =>
            Store(useMetica ? global::MainAdNetwork.Metica : global::MainAdNetwork.MAX, null);

        /// <summary>
        /// Reads the switch out of an AdsSettings JSON — the Firebase parameter GD SDK games
        /// use, {"MainAdNetwork":"Metica"} — and applies it as <see cref="Apply(string)"/> does.
        /// Without that key the earlier {"UseMetica":true} still counts; with neither, nothing
        /// changes.
        /// </summary>
        /// <returns>True when a value was stored.</returns>
        public static bool ApplyAdsSettings(string adsSettingsJson)
        {
            AdsSettings settings = null;
            try
            {
                if (!string.IsNullOrEmpty(adsSettingsJson)) settings = JsonUtility.FromJson<AdsSettings>(adsSettingsJson);
            }
            catch (ArgumentException)
            {
                MeticaAdsLog.LogError($"Remote AdsSettings is not valid JSON, MainAdNetwork stays {MainAdNetwork}");
                return false;
            }

            if (settings != null && !string.IsNullOrWhiteSpace(settings.MainAdNetwork))
                return Apply(settings.MainAdNetwork);

            // The earlier key. JsonUtility hands a JSON bool to a string field as "true" / "false".
            var useMetica = settings?.UseMetica?.Trim().ToLowerInvariant();
            if (useMetica == "true" || useMetica == "1" || useMetica == "false" || useMetica == "0")
            {
                Store(useMetica == "true" || useMetica == "1" ? global::MainAdNetwork.Metica : global::MainAdNetwork.MAX,
                    " (from the earlier UseMetica key)");
                return true;
            }

            if (!string.IsNullOrEmpty(useMetica))
            {
                MeticaAdsLog.LogError($"Remote UseMetica \"{settings.UseMetica}\" is not true or false, " +
                                      $"MainAdNetwork stays {MainAdNetwork}");
                return false;
            }

            MeticaAdsLog.Log($"Remote AdsSettings has no MainAdNetwork, MainAdNetwork stays {MainAdNetwork}");
            return false;
        }

        /// <summary>The two fields of AdsSettings this reads; a fresh one per fetch, so a missing key reads as null.</summary>
        [Serializable]
        private sealed class AdsSettings
        {
            public string MainAdNetwork = null;
            public string UseMetica = null;
        }

        private static void Store(MainAdNetwork value, string note)
        {
            var changed = MainAdNetwork != value;

            MainAdNetworkPref.Set(value);

            MeticaAdsLog.Log((changed
                ? $"Remote switch stored: MainAdNetwork = {value}"
                : $"Remote switch unchanged: MainAdNetwork = {value}") + note);
        }

        /// <summary>Forgets the stored value, so the next session is Unassigned again.</summary>
        public static void Clear()
        {
            PlayerPrefs.DeleteKey(MainAdNetworkPref.Key);
            PlayerPrefs.Save();
            MeticaAdsLog.Log("Remote switch cleared");
        }

        /// <summary>
        /// Whether Metica should come up this session. With the switch off this is always
        /// true; with it on it is the network stored last session, or the config's default
        /// while that is Unassigned.
        ///
        /// <para>The first call in a session logs the saved pref ("MainAdNetwork = Metica from
        /// saved pref → Metica this session") and reports the network, through
        /// <see cref="MeticaAdsHooks.OnEvent"/>: "GameData" with { "MainAdNetwork": "Unassigned" /
        /// "MAX" / "Metica" } — the same event the GD Monetization SDK sends.</para>
        /// </summary>
        public static bool ShouldInitialize(MeticaAdsConfig config)
        {
            bool useMetica;
            MainAdNetwork reported;
            if (config == null || !config.UseRemoteSwitch)
            {
                useMetica = true;
                reported = global::MainAdNetwork.Metica;
            }
            else
            {
                reported = MainAdNetwork;
                useMetica = reported == global::MainAdNetwork.Unassigned
                    ? config.DefaultUseMetica
                    : reported == global::MainAdNetwork.Metica;
            }

            // Once per session, at ads init: the saved pref as stored (Unassigned, MAX or
            // Metica), and the network it gives this session. A warning, so it prints with
            // logging off too.
            if (!_logged)
            {
                _logged = true;
                MeticaAdsLog.LogWarning($"MainAdNetwork = {MainAdNetwork} from saved pref → " +
                                        (useMetica ? "Metica" : "AppLovin") + " this session" +
                                        (config == null || !config.UseRemoteSwitch ? " (remote switch off: Metica always)" : ""));
            }

            // Wire MeticaAdsHooks.OnEvent before ads start (BeforeSceneLoad), or this one is missed.
            if (!_reported && MeticaAdsHooks.OnEvent != null)
            {
                _reported = true;
                MeticaAdsHooks.ReportEvent("GameData", new Dictionary<string, object> { { "MainAdNetwork", reported.ToString() } });
            }

            return useMetica;
        }
    }
}
