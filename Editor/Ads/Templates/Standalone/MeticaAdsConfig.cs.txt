using UnityEngine;

namespace MeticaIntegration
{
    /// <summary>
    /// Every id Metica ads need, in one asset. Nothing here is code — it is the one place a
    /// developer edits, and the one place to look when an ad will not serve.
    ///
    /// <para>Lives at <c>Assets/Resources/MeticaAdsConfig.asset</c> because it is loaded by
    /// name at runtime. Ad unit ids differ per platform, so each has an Android and an iOS
    /// field, and <see cref="Interstitial"/> and friends resolve against the build target.</para>
    /// </summary>
    [CreateAssetMenu(fileName = AssetName, menuName = "Metica/Ads Config")]
    public sealed class MeticaAdsConfig : ScriptableObject
    {
        public const string AssetName = "MeticaAdsConfig";

        [Header("Metica")]
        [Tooltip("Metica App ID, from the Metica dashboard.")]
        public string AppId;

        [Tooltip("Metica API key, from the Metica dashboard.")]
        public string ApiKey;

        [Header("AppLovin MAX")]
        [Tooltip("MAX SDK key. Metica serves through MAX, so this is your existing MAX key.")]
        public string MaxSdkKey;

        [Header("Ad unit ids — Android")]
        public string AndroidInterstitial;
        public string AndroidRewarded;
        public string AndroidBanner;
        public string AndroidMRec;

        [Header("Ad unit ids — iOS")]
        public string IosInterstitial;
        public string IosRewarded;
        public string IosBanner;
        public string IosMRec;

        [Header("Ad positions")]
        [Tooltip("Where the banner appears when it is created. Bottom by default. " +
                 "MeticaAdsManager.RepositionBanner moves it at runtime.")]
        public BannerPosition BannerPosition = BannerPosition.Bottom;

        [Tooltip("Where the MREC appears when it is created. Center by default. " +
                 "MeticaAdsManager.RepositionMRec moves it at runtime.")]
        public MRecPosition MRecPosition = MRecPosition.Center;

        [Tooltip("Shift the MREC from MRec Position, in dp (points on iOS): x right, y down. " +
                 "Bottom with (0, -140) puts it 140 dp above the bottom edge. (0, 0): no shift. " +
                 "An MREC is always 300 × 250 — Metica cannot resize it.")]
        public Vector2 MRecOffset;

        [Header("Remote switch")]
        [Tooltip("Gate Metica on a value from your own remote config. On (the default): Metica stays " +
                 "off until the backend turns it on through MeticaRemoteConfig.Apply(true). Unticked: " +
                 "Metica is always on, and MeticaRemoteConfig is never consulted.")]
        public bool UseRemoteSwitch = true;

        [Tooltip("What the switch reads until your remote config has landed once on this device. " +
                 "Off (the default): Metica stays off until the backend turns it on.")]
        public bool DefaultUseMetica;

        [Header("Behaviour")]
        [Tooltip("Print Metica's own SDK logs. On by default while integrating; turn off for release.")]
        public bool MeticaSdkLogging = true;

        [Tooltip("Print this integration's logs. Warnings and errors print either way.")]
        public bool Logging = true;

        // ── The ids for the platform actually being built ──────────────────────

        // Chosen by build target rather than RuntimePlatform, so the editor uses the ids of
        // the platform you are actually building for.
#if UNITY_IOS
        public string Interstitial => IosInterstitial;
        public string Rewarded => IosRewarded;
        public string Banner => IosBanner;
        public string MRec => IosMRec;
#else
        public string Interstitial => AndroidInterstitial;
        public string Rewarded => AndroidRewarded;
        public string Banner => AndroidBanner;
        public string MRec => AndroidMRec;
#endif

        /// <summary>Loads the asset from Resources. Null when it has not been created yet.</summary>
        public static MeticaAdsConfig Load() => Resources.Load<MeticaAdsConfig>(AssetName);

        /// <summary>
        /// The ids that must be present for initialization to be worth attempting. Ad unit
        /// ids are not included — a game with no banner is a perfectly ordinary game.
        /// </summary>
        public bool IsUsable(out string problem)
        {
            if (string.IsNullOrEmpty(ApiKey)) { problem = "ApiKey is empty"; return false; }
            if (string.IsNullOrEmpty(AppId)) { problem = "AppId is empty"; return false; }
            if (string.IsNullOrEmpty(MaxSdkKey)) { problem = "MaxSdkKey is empty"; return false; }

            problem = null;
            return true;
        }
    }
}
