using System;
using System.Collections.Generic;

namespace MeticaIntegration
{
    /// <summary>
    /// Where this integration meets the rest of your game.
    ///
    /// <para>Everything else here is self-contained and needs no editing. These four hooks
    /// are the exception, because only your project knows what its analytics and consent
    /// look like. All four are optional: leave them unset and ads still serve — you simply
    /// get no revenue events and Metica is told nothing about consent.</para>
    ///
    /// <para>Set them once, before <c>MeticaAdsManager.Initialize()</c>. A
    /// <c>RuntimeInitializeOnLoadMethod</c> or your existing bootstrap is the usual place.</para>
    /// </summary>
    public static class MeticaAdsHooks
    {
        /// <summary>
        /// Called for every ad impression that earns revenue. Forward it to whatever ad-revenue
        /// reporting you already have.
        ///
        /// <para><b>Not always on the Unity main thread.</b> Interstitial and rewarded revenue
        /// arrive on Metica's native thread the moment the ad pays — while the ad still has Unity
        /// paused — so the event is sent even if the app is killed before the ad closes. Banner
        /// and MREC revenue arrive on the main thread.</para>
        ///
        /// <para>So in this handler: analytics SDK calls are fine (Firebase LogEvent, Adjust,
        /// AppMetrica — they go straight to native code). Anything that touches Unity —
        /// PlayerPrefs, MonoBehaviours, GameObjects, Time — goes through
        /// <c>ThreadDispatcher.Enqueue(() => …)</c>, which runs it on the main thread once
        /// Unity resumes.</para>
        /// </summary>
        public static Action<AdRevenueInfo> OnAdRevenue;

        /// <summary>
        /// Whether the user has consented to personalized ads. Called once, just after
        /// Metica initializes. Unset is treated as no consent, which is the safe default.
        /// </summary>
        public static Func<bool> HasUserConsent;

        /// <summary>
        /// A named event with parameters, for your analytics. Used once, to report the
        /// Smart Floors user group and the Metica SDK version after initialization.
        /// </summary>
        public static Action<string, IDictionary<string, object>> OnEvent;

        /// <summary>
        /// The user clicked an ad and is probably about to leave the app. If you have an
        /// app-open ad or a session timer, this is where to hold it off.
        /// </summary>
        public static Action OnAdClicked;

        // ── Example wiring ─────────────────────────────────────────────────────
        //
        // Put something like this in your own bootstrap — not in this file, so it survives
        // the tool being run again:
        //
        //  [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.BeforeSceneLoad)]
        //  private static void WireMeticaAds()
        //  {
        //      MeticaAdsHooks.OnAdRevenue = info =>
        //      {
        //          // Native thread for interstitial / rewarded: SDK calls only here…
        //          MyAnalytics.LogAdRevenue(info.NetworkName, info.Revenue, info.Currency);
        //          // …and Unity work on the main thread.
        //          ThreadDispatcher.Enqueue(() => MyRevenueCounter.Add(info.Revenue));
        //      };
        //
        //      MeticaAdsHooks.HasUserConsent = () => MyConsent.PersonalizedAdsAllowed;
        //
        //      MeticaAdsHooks.OnEvent = (name, parameters) =>
        //          MyAnalytics.LogEvent(name, parameters);
        //
        //      MeticaAdsHooks.OnAdClicked = () => MyAppOpenAd.PostponeNextShow();
        //  }

        // ── Called by the integration ──────────────────────────────────────────

        /// <summary>A hook belongs to the game, so it must never take the ads stack down with it.</summary>
        internal static void ReportRevenue(AdRevenueInfo info)
        {
            if (OnAdRevenue == null) return;

            try { OnAdRevenue(info); }
            catch (Exception e) { MeticaAdsLog.LogError($"OnAdRevenue threw: {e}"); }
        }

        internal static void ReportEvent(string name, IDictionary<string, object> parameters)
        {
            if (OnEvent == null) return;

            try { OnEvent(name, parameters); }
            catch (Exception e) { MeticaAdsLog.LogError($"OnEvent threw: {e}"); }
        }

        internal static void ReportAdClicked()
        {
            if (OnAdClicked == null) return;

            try { OnAdClicked(); }
            catch (Exception e) { MeticaAdsLog.LogError($"OnAdClicked threw: {e}"); }
        }

        internal static bool ReadConsent()
        {
            if (HasUserConsent == null) return false;

            try { return HasUserConsent(); }
            catch (Exception e)
            {
                MeticaAdsLog.LogError($"HasUserConsent threw: {e}");
                return false;
            }
        }
    }
}
