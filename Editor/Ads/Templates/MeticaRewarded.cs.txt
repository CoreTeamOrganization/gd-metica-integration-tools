using System;
using Metica;
using Metica.Ads;
using Monetization.Runtime.Analytics;
using Monetization.Runtime.Logger;
using Monetization.Runtime.Utilities;

namespace Monetization.Runtime.Ads
{
    internal sealed class MeticaRewarded : RewardedAdUnit
    {
        private int retryAttempt;
        private bool requestAdOnFail = true;


        public override void Initialize(string adUnitId)
        {
            this.adUnitId = adUnitId;
            if (IsAdUnitEmpty)
                return;

            ListenEvents();
            LoadRewarded();
        }

        private void ListenEvents()
        {
            MeticaAdsCallbacks.Rewarded.OnAdHidden += OnRewardedHiddenEvent;
            MeticaAdsCallbacks.Rewarded.OnAdLoadFailed += OnRewardedLoadFailedEvent;
            MeticaAdsCallbacks.Rewarded.OnAdRevenuePaid += OnRewardedRevenuePaidEvent;
            MeticaAdsCallbacks.Rewarded.OnAdRewarded += OnRewardedRewardedEvent;

            MeticaAdsCallbacks.Rewarded.OnAdLoadSuccess += OnRewardedLoadedEvent;
            MeticaAdsCallbacks.Rewarded.OnAdShowFailed += OnRewardedShowFailedEvent;
        }

        public override bool HasRewarded(bool doRequest)
        {
            if (IsAdUnitEmpty)
                return false;

            bool isReady = MeticaSdk.Ads.IsRewardedReady(adUnitId);
            if (!isReady && doRequest)
                LoadRewarded();

            return isReady;
        }

        public override void ShowRewarded(string placementName, Action onRewardedCallback)
        {
            this.placementName = placementName;
            this.onRewardComplete = onRewardedCallback;
            MeticaSdk.Ads.ShowRewarded(adUnitId, placementName);
        }

        public override void LoadRewarded()
        {
            if (IsAdUnitEmpty)
                return;

            MeticaSdk.Ads.LoadRewarded(adUnitId);
        }

        #region Callbacks

        private void OnRewardedLoadedEvent(MeticaAd meticaAd)
        {
            retryAttempt = 0;
            Message.Log(Tag.Metica, "Rewarded ad loaded");
        }

        private void OnRewardedLoadFailedEvent(MeticaAdError meticaAdError)
        {
            if (requestAdOnFail)
            {
                requestAdOnFail = false;
                retryAttempt++;
                float retryDelay = UnityEngine.Mathf.Pow(2, UnityEngine.Mathf.Min(6, retryAttempt));
                DelayedActionManager.Add(LoadRewarded, retryDelay);
                Message.LogWarning(Tag.Metica,
                    $"Rewarded ad failed to load. Retrying in {retryDelay:0.##}s");
            }
        }

        private void OnRewardedShowFailedEvent(MeticaAd meticaAd, MeticaAdError meticaAdError)
        {
            ThreadDispatcher.Enqueue(LoadRewarded);
        }

        private void OnRewardedHiddenEvent(MeticaAd meticaAd)
        {
            ThreadDispatcher.Enqueue(LoadRewarded);
            Message.Log(Tag.Metica, "Rewarded_Ad hidden");
        }

        private void OnRewardedRewardedEvent(MeticaAd meticaAd)
        {
            ThreadDispatcher.Enqueue(() => { onRewardComplete?.Invoke(); });
            Message.Log(Tag.Metica, "Rewarded-Ad-Reward rewarded");
        }

        private void OnRewardedRevenuePaidEvent(MeticaAd meticaAd)
        {
            double revenue = meticaAd.revenue ?? 0.0;
            string networkName = meticaAd.networkName ?? "Unknown";
            string adUnitIdentifier = meticaAd.adUnitId;
            string placement = meticaAd.placementTag ?? placementName;

            Message.Log(Tag.Metica,
                $"Rewarded revenue paid - AdUnit: {adUnitIdentifier}, Revenue: {revenue}, Network: {networkName}, Placement: {placement}");

            AdRevenueInfo revInfo = new(AdFormat.Rewarded, AdPlatforms.METICA, networkName,
                adUnitIdentifier, "USD", revenue, placement, AdNetworkMetica.RevenuePayload);

            AnalyticsManager.ReportAdRevenue(revInfo);
        }

        #endregion
    }
}
